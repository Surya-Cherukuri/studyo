Median of Two Sorted Arrays · Hard

Two sorted lists; return the median of all their numbers together, in O(log(m + n)) time.

▶ Watch the visual lesson

Binary search on the cut

O(log(min(m, n))) time · O(1) space

Cut both lists so the left halves hold half of all numbers. The cut is right when the four numbers around it are in order; binary search the cut in the shorter list.

from typing import List
from math import inf
class Solution:
    def findMedianSortedArrays(self,
            nums1: List[int],
            nums2: List[int]) -> float:
        a, b = nums1, nums2
        if len(a) > len(b):
            a, b = b, a
        m, n = len(a), len(b)
        half = (m + n + 1) // 2
        lo, hi = 0, m
        while lo <= hi:
            i = (lo + hi) // 2
            j = half - i
            a_left = a[i - 1] if i > 0 else -inf
            a_right = a[i] if i < m else inf
            b_left = b[j - 1] if j > 0 else -inf
            b_right = b[j] if j < n else inf
            if a_left <= b_right and b_left <= a_right:
                left = max(a_left, b_left)
                if (m + n) % 2 == 1:
                    return left
                right = min(a_right, b_right)
                return (left + right) / 2
            elif a_left > b_right:
                hi = i - 1
            else:
                lo = i + 1
